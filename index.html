async function sendPhoto(input) { const file = input.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = async function () { const image = reader.result; const chat = document.getElementById("chat"); const photoBox = document.createElement("div"); photoBox.className = "msg user"; const photoImg = document.createElement("img"); photoImg.src = image; photoImg.style.maxWidth = "220px"; photoImg.style.borderRadius = "12px"; photoImg.style.display = "block"; photoBox.appendChild(photoImg); chat.appendChild(photoBox); chat.scrollTop = chat.scrollHeight; try { const response = await fetch( "https://chat-sk.onrender.com/vision", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ image: image, question: "या फोटोमध्ये काय आहे ते समजावून सांग." }) } ); const data = await response.json(); addMessage(data.reply || "उत्तर मिळाले नाही.", "bot"); } catch (error) { addMessage( "फोटो AI कडे पाठवताना समस्या आली.", "bot" ); } }; reader.readAsDataURL(file); }
